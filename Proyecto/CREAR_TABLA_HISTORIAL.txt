CREATE TABLE historial_entregas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    solicitud_id INT NOT NULL,
    mecanico_id INT NOT NULL,
    repuesto_id INT NOT NULL,
    proveedor_id INT NOT NULL,
    cantidad INT UNSIGNED NOT NULL CHECK (cantidad > 0),
    fecha_solicitud DATE NOT NULL,
    fecha_entrega DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    estado_anterior ENUM('APROBADA','EN_PREPARACION','ENVIADO','ENTREGADO','CANCELADO') NOT NULL,
    notas TEXT NULL,
    entregado_por VARCHAR(100) NULL,
    recibido_por VARCHAR(100) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_solicitud (solicitud_id),
    INDEX idx_mecanico (mecanico_id),
    INDEX idx_repuesto (repuesto_id),
    INDEX idx_proveedor (proveedor_id),
    INDEX idx_fecha_entrega (fecha_entrega),
    
    FOREIGN KEY (solicitud_id) REFERENCES solicitudes(id) ON DELETE CASCADE,
    FOREIGN KEY (mecanico_id) REFERENCES mecanicos(id) ON DELETE CASCADE,
    FOREIGN KEY (repuesto_id) REFERENCES repuestos(id) ON DELETE CASCADE,
    FOREIGN KEY (proveedor_id) REFERENCES proveedores(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
